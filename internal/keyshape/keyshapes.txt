# Key SHAPES, matched against text a worker produced.
#
# This file is DATA, and what is in it is a SHAPE and never a value. The reader holds no
# key and reads none: it cannot compare a line against "our key", because having our key
# to compare against is the thing that must not be true of the machine that reads every
# worker's output. What it can do is recognise the form a key takes, which is public
# knowledge published by each issuer.
#
# A match is reported as `shape=<name> file=<path> line=<n>`, and the matched text is
# NEVER printed: a finding that quotes the key has copied the key into every log that
# carries the finding.
#
# This list is kept byte-identical with internal/hygiene/keyshapes.txt where the two
# overlap (the hygiene gate, nova-tools T02, scans a card's added diff lines with the same
# shapes); when that gate lands, one of the two loads the other and neither is copied
# again. The two rows below it does not yet carry are marked.
#
# Format: <name><TAB><Go regular expression>

# Private keys in PEM armour. The header is the shape; the body is never looked at.
pem-private-key	-----BEGIN [A-Z0-9 ]*PRIVATE KEY-----

# age, which this fleet uses for at-rest secrets.
age-secret-key	AGE-SECRET-KEY-1[0-9A-Za-z]{10,}

# The forge's own token prefixes, published by GitHub.
forge-token	gh[pousr]_[A-Za-z0-9]{20,}
forge-fine-grained-token	github_pat_[A-Za-z0-9_]{20,}

# Provider API keys, by their published prefixes. The bare `sk-` shape is {20,} rather
# than hygiene's {32,}: the key this fleet's seats actually hold is `sk-` and 32 more
# (measured length 35, issue #1814), and a shorter bound catches a truncated copy of it
# without reaching prose, which does not carry twenty unbroken alphanumerics after `sk-`.
anthropic-api-key	sk-ant-[A-Za-z0-9_-]{20,}
openai-api-key	sk-[A-Za-z0-9]{20,}
xai-api-key	xai-[A-Za-z0-9-]{20,}
google-api-key	AIza[0-9A-Za-z_-]{30,}
slack-token	xox[abprs]-[0-9A-Za-z-]{10,}
aws-access-key-id	(A3T[A-Z0-9]|AKIA|ASIA|ABIA|ACCA)[A-Z0-9]{16}

# The generic shapes the issuers do not prefix: a private key blob in an ssh config, and
# a JSON web token. Both are deliberately narrow; a shape that matches prose would make
# every card a secret finding and the check would be turned off within a week.
ssh-private-key-blob	PRIVATE KEY BLOCK-----
json-web-token	eyJ[A-Za-z0-9_-]{10,}\.eyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}
